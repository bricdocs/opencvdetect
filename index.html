function canliTaramaDongusu() { if (!opencvHazir || !streamActive) return; // Kamerayı her karede ekranda göstermeye devam et ctx.drawImage(video, 0, 0, 640, 480); // -------------------------------------------------- // Telefonu rahatlat: // OpenCV taraması yaklaşık 8 FPS // -------------------------------------------------- const simdi = performance.now(); if ( typeof canliTaramaDongusu.sonTarama !== "undefined" && simdi - canliTaramaDongusu.sonTarama < 125 ) { requestAnimationFrame(canliTaramaDongusu); return; } canliTaramaDongusu.sonTarama = simdi; try { // -------------------------------------------------- // 1. Kameradan TAM 640x480 görüntü al // -------------------------------------------------- let imgData = ctx.getImageData(0, 0, 640, 480); let src = cv.matFromImageData(imgData); // -------------------------------------------------- // 2. GRAY görüntü // -------------------------------------------------- let gray = new cv.Mat(); cv.cvtColor( src, gray, cv.COLOR_RGBA2GRAY ); let kartBulundu = false; let enIyiKart = ""; let enIyiSkor = 0; let enIyiLoc = null; let enIyiBoyut = { w: 0, h: 0 }; // -------------------------------------------------- // 3. Bütün kart şablonlarını tara // -------------------------------------------------- for (let isim in sablonMatrisleri) { let sablon = sablonMatrisleri[isim]; let result = new cv.Mat(); cv.matchTemplate( gray, sablon, result, cv.TM_CCOEFF_NORMED ); let minMax = cv.minMaxLoc(result); // En yüksek skoru sakla if (minMax.maxVal > enIyiSkor) { enIyiSkor = minMax.maxVal; enIyiKart = isim; enIyiLoc = minMax.maxLoc; enIyiBoyut = { w: sablon.cols, h: sablon.rows }; } result.delete(); } // -------------------------------------------------- // 4. Eşik kontrolü // -------------------------------------------------- if (enIyiSkor >= 0.50) { kartBulundu = true; resultText.innerText = `KART ALGILANDI: ${enIyiKart} (%${Math.round(enIyiSkor * 100)})`; resultText.style.color = "#00ff00"; // -------------------------------------------------- // 5. YEŞİL DİKDÖRTGEN // // Görüntü yine 640x480 olduğu için // koordinat dönüşümü YOK. // -------------------------------------------------- ctx.beginPath(); ctx.lineWidth = 4; ctx.strokeStyle = "#00ff00"; ctx.rect( enIyiLoc.x, enIyiLoc.y, enIyiBoyut.w, enIyiBoyut.h ); ctx.stroke(); } else { resultText.innerText = `Kart aranıyor... En iyi skor: %${Math.round(enIyiSkor * 100)}`; resultText.style.color = "#fff"; } // -------------------------------------------------- // 6. Bellek temizliği // -------------------------------------------------- src.delete(); gray.delete(); } catch (err) { console.error("Döngü hatası: ", err); } requestAnimationFrame(canliTaramaDongusu); }
