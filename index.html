function canliTaramaDongusu() { if (!opencvHazir || !streamActive) return; // Kamerayı ekranda göstermeye devam et ctx.drawImage(video, 0, 0, 640, 480); // OpenCV taraması yaklaşık 8 FPS const simdi = performance.now(); if ( typeof canliTaramaDongusu.sonTarama !== "undefined" && simdi - canliTaramaDongusu.sonTarama < 125 ) { requestAnimationFrame(canliTaramaDongusu); return; } canliTaramaDongusu.sonTarama = simdi; try { let src; // ========================================== // 1. KAMERA GÖRÜNTÜSÜ // ========================================== if (video.videoWidth < video.videoHeight) { // TELEFON: // Gerçek görüntü: 480 x 640 // Ortadan 480 x 480 kırp let telefonCanvas = document.createElement("canvas"); telefonCanvas.width = video.videoWidth; telefonCanvas.height = video.videoWidth; let telefonCtx = telefonCanvas.getContext("2d"); // Dikey görüntünün ortasındaki 480x480 alanı al const kirpY = Math.round((video.videoHeight - video.videoWidth) / 2); telefonCtx.drawImage( video, 0, kirpY, video.videoWidth, video.videoWidth, 0, 0, video.videoWidth, video.videoWidth ); let telefonImgData = telefonCtx.getImageData( 0, 0, video.videoWidth, video.videoWidth ); src = cv.matFromImageData(telefonImgData); } else { // PC: // Mevcut 640x480 sistemi aynen koru let imgData = ctx.getImageData( 0, 0, 640, 480 ); src = cv.matFromImageData(imgData); } // ========================================== // 2. GRAY // ========================================== let gray = new cv.Mat(); cv.cvtColor( src, gray, cv.COLOR_RGBA2GRAY ); // ========================================== // 3. EN İYİ KARTI BUL // ========================================== let enIyiKart = ""; let enIyiSkor = 0; let enIyiLoc = null; let enIyiBoyut = { w: 0, h: 0 }; for (let isim in sablonMatrisleri) { let sablon = sablonMatrisleri[isim]; let result = new cv.Mat(); cv.matchTemplate( gray, sablon, result, cv.TM_CCOEFF_NORMED ); let minMax = cv.minMaxLoc(result); if (minMax.maxVal > enIyiSkor) { enIyiSkor = minMax.maxVal; enIyiKart = isim; enIyiLoc = minMax.maxLoc; enIyiBoyut = { w: sablon.cols, h: sablon.rows }; } result.delete(); } // ========================================== // 4. EŞİK // ========================================== if (enIyiSkor >= 0.50) { resultText.innerText = `KART ALGILANDI: ${enIyiKart} (%${Math.round(enIyiSkor * 100)})`; resultText.style.color = "#00ff00"; // Şimdilik kutuyu çizme. // Çünkü telefon görüntüsü 480x480, // ekran görüntüsü 640x480. // Önce skorun davranışını test ediyoruz. } else { resultText.innerText = `Kart aranıyor... En iyi skor: %${Math.round(enIyiSkor * 100)} (${enIyiKart || "yok"}) | Kamera: ${video.videoWidth}x${video.videoHeight}`; resultText.style.color = "#fff"; } src.delete(); gray.delete(); } catch (err) { console.error( "Döngü hatası: ", err ); } requestAnimationFrame(canliTaramaDongusu); }
